cmake_minimum_required(VERSION 3.18)
project(v16_wbc_native LANGUAGES CXX)
set(PYBIND11_FINDPYTHON ON)
find_package(Python3 REQUIRED COMPONENTS Interpreter Development.Module)
find_package(pybind11 CONFIG REQUIRED)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
pybind11_add_module(_v16_wbc_native wbc_native.cpp)
target_link_libraries(_v16_wbc_native PRIVATE Eigen3::Eigen)
target_compile_features(_v16_wbc_native PRIVATE cxx_std_17)
target_compile_options(_v16_wbc_native PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
